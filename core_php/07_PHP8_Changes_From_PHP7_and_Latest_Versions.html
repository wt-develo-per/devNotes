Interactive PHP 8.0 → 8.4 Version Timeline & Feature Matrix

Step through the rapid evolution of PHP across every major 8.x release, exploring paradigm shifts from 8.0 to cutting-edge 8.4:

PHP 8.4 State-of-the-Art: Property Hooks (get/set) and Asymmetric Visibility (public private(set)).

<?php
declare(strict_types=1);

class UserAccount
{
    // Asymmetric Visibility: Anyone can read, but only internal methods can mutate!
    public private(set) int $loginAttempts = 0;

    // Property Hooks: Eliminate trivial getter/setter boilerplate!
    public string $email {
        set {
            if (!filter_var($value, FILTER_VALIDATE_EMAIL)) {
                throw new \InvalidArgumentException('Invalid email format');
            }
            $this->email = strtolower(trim($value));
        }
        get => $this->email;
    }

    // Computed Virtual Property (No backing storage needed):
    public string $initials {
        get => strtoupper(substr($this->firstName, 0, 1) . substr($this->lastName, 0, 1));
    }

    public function __construct(
        public string $firstName,
        public string $lastName,
        string $email
    ) {
        $this->email = $email;
    }

    public function recordFailedLogin(): void
    {
        $this->loginAttempts++; // Allowed internally!
    }
}

$user = new UserAccount('John', 'Doe', ' JOHN@DOMAIN.COM ');
echo $user->email;            // 'john@domain.com'
echo $user->initials;         // 'JD'
echo $user->loginAttempts;    // 0
// $user->loginAttempts = 5;  // Fatal Error: Cannot modify private(set) property from global scope!
?>

PHP 8 Native Attributes: Type-safe, compile-time metadata replacing doctrine docblock comments.

<?php
declare(strict_types=1);

namespace App\Attributes;

use Attribute;

#[Attribute(Attribute::TARGET_METHOD | Attribute::TARGET_FUNCTION)]
class Route
{
    public function __construct(
        public string $path,
        public string $method = 'GET'
    ) {}
}

// Controller using Native Attributes (Modern Framework Pattern):
class UserController
{
    #[Route('/api/v1/users', method: 'GET')]
    public function index(): array
    {
        return ['status' => 'active users list'];
    }

    #[Route('/api/v1/users', method: 'POST')]
    public function store(): array
    {
        return ['status' => 'user created'];
    }
}

// Reflection Engine reading attributes at runtime:
$reflector = new ReflectionClass(UserController::class);
foreach ($reflector->getMethods() as $method) {
    $attributes = $method->getAttributes(Route::class);
    foreach ($attributes as $attribute) {
        $routeInstance = $attribute->newInstance();
        echo "Found Route: {$routeInstance->method} {$routeInstance->path} => {$method->getName()}\n";
    }
}
?>

Critical Migration Pitfall: String-to-Number Comparison Logic completely inverted in PHP 8!

<?php
// ==================== THE PHP 7 DANGER ====================
// In PHP 7: When comparing number to string, PHP cast the STRING to a NUMBER!
// 'foobar' cast to integer became 0!
0 == 'foobar'; // TRUE in PHP 7! (Catastrophic bug in authentication / authorization!)
0 == '';       // TRUE in PHP 7!
42 == '42foo'; // TRUE in PHP 7!

// ==================== THE PHP 8 FIX =======================
// In PHP 8: When comparing number to non-numeric string, PHP converts NUMBER to STRING!
0 == 'foobar'; // FALSE in PHP 8! ('0' == 'foobar')
0 == '';       // FALSE in PHP 8! ('0' == '')
42 == '42foo'; // FALSE in PHP 8! (Leading numeric string comparison deprecated/removed)
0 == '0';      // TRUE in PHP 8! (Both represent numeric string '0')

// ==================== SWITCH vs MATCH =====================
// switch statement uses LOOSE COMPARISON (==):
switch (0) {
    case 'admin': // In PHP 7, MATCHED! Admin access granted! In PHP 8, does not match.
        break;
}

// match expression uses STRICT TYPE CHECKING (===):
$role = match(0) {
    'admin' => 'Administrator', // Will NEVER match integer 0!
    0       => 'Guest User',
};
?>

PHP 8.4 Array Family & Lazy Objects: Native performance without third-party libraries.

<?php
declare(strict_types=1);

// 1. PHP 8.4 Functional Array Helpers:
$users = [
    ['id' => 1, 'name' => 'Alice', 'role' => 'subscriber', 'active' => true],
    ['id' => 2, 'name' => 'Bob',   'role' => 'editor',     'active' => true],
    ['id' => 3, 'name' => 'Carol', 'role' => 'admin',      'active' => false],
];

// Find first element matching predicate:
$admin = array_find($users, fn($u) => $u['role'] === 'admin');
// Result: ['id' => 3, 'name' => 'Carol', ...]

// Check if any element matches predicate:
$hasAdmin = array_any($users, fn($u) => $u['role'] === 'admin'); // true

// Check if all elements match predicate:
$allActive = array_all($users, fn($u) => $u['active'] === true); // false

// 2. PHP 8.4 Native Lazy Objects (Ghost Objects):
// Defers heavy database hydration until a property or method is actually accessed!
$reflector = new ReflectionClass(HeavyDatabaseModel::class);
$lazyInstance = $reflector->newLazyGhost(function (HeavyDatabaseModel $ghost) {
    // Only runs when $lazyInstance->someProperty is touched!
    $ghost->hydrateFromRemoteDatabase();
});
?>
