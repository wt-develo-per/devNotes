Interactive REST API Request-Response Architecture Pipeline

Follow an incoming HTTP request step-by-step through a production-ready Core PHP API engine without frameworks:

Custom Regex Router: Dynamic parameter extraction (/api/v1/users/{id}) and HTTP method dispatching.

<?php
declare(strict_types=1);

namespace App\Core;

class Router
{
    private array $routes = [];

    public function get(string $path, array|callable $handler, array $middlewares = []): void
    {
        $this->addRoute('GET', $path, $handler, $middlewares);
    }

    public function post(string $path, array|callable $handler, array $middlewares = []): void
    {
        $this->addRoute('POST', $path, $handler, $middlewares);
    }

    public function addRoute(string $method, string $path, array|callable $handler, array $middlewares): void
    {
        // Convert route pattern /users/{id} to regex #^/users/(?P<id>[^/]+)$#
        $pattern = preg_replace('#\{([a-zA-Z0-9_]+)\}#', '(?P<$1>[^/]+)', $path);
        $regex = '#^' . $pattern . '$#';

        $this->routes[] = [
            'method'      => strtoupper($method),
            'pattern'     => $path,
            'regex'       => $regex,
            'handler'     => $handler,
            'middlewares' => $middlewares,
        ];
    }

    public function dispatch(string $requestMethod, string $requestUri): void
    {
        $path = parse_url($requestUri, PHP_URL_PATH) ?? '/';

        foreach ($this->routes as $route) {
            if ($route['method'] !== $requestMethod) {
                continue;
            }

            if (preg_match($route['regex'], $path, $matches)) {
                // Filter named captures from regex matches
                $params = array_filter($matches, 'is_string', ARRAY_FILTER_USE_KEY);

                // Run Middleware Pipeline
                foreach ($route['middlewares'] as $middlewareClass) {
                    $middleware = new $middlewareClass();
                    $middleware->handle(); // Throws exception if unauthorized
                }

                // Execute Controller Handler
                if (is_array($route['handler'])) {
                    [$class, $method] = $route['handler'];
                    $controller = new $class();
                    $response = $controller->{$method}($params);
                } else {
                    $response = call_user_func($route['handler'], $params);
                }

                Response::json($response);
                return;
            }
        }

        Response::json(['error' => 'Route Not Found', 'path' => $path], 404);
    }
}
?>

JWT Authentication from Scratch: Pure PHP implementation without external dependencies.

<?php
declare(strict_types=1);

namespace App\Core;

class SimpleJWT
{
    private static function base64UrlEncode(string $data): string
    {
        return rtrim(strtr(base64_encode($data), '+/', '-_'), '=');
    }

    private static function base64UrlDecode(string $data): string
    {
        return base64_decode(strtr($data, '-_', '+/'));
    }

    public static function encode(array $payload, string $secret, int $expirySeconds = 3600): string
    {
        $header = ['typ' => 'JWT', 'alg' => 'HS256'];
        $payload['iat'] = time();
        $payload['exp'] = time() + $expirySeconds;

        $encodedHeader  = self::base64UrlEncode(json_encode($header, JSON_THROW_ON_ERROR));
        $encodedPayload = self::base64UrlEncode(json_encode($payload, JSON_THROW_ON_ERROR));

        $signature = hash_hmac('sha256', "{$encodedHeader}.{$encodedPayload}", $secret, true);
        $encodedSignature = self::base64UrlEncode($signature);

        return "{$encodedHeader}.{$encodedPayload}.{$encodedSignature}";
    }

    public static function decode(string $token, string $secret): array
    {
        $parts = explode('.', $token);
        if (count($parts) !== 3) {
            throw new \InvalidArgumentException('Malformed JWT token structure.');
        }

        [$encodedHeader, $encodedPayload, $encodedSignature] = $parts;

        // Verify HMAC-SHA256 signature in constant time
        $expectedSignature = self::base64UrlEncode(
            hash_hmac('sha256', "{$encodedHeader}.{$encodedPayload}", $secret, true)
        );

        if (!hash_equals($expectedSignature, $encodedSignature)) {
            throw new \RuntimeException('Invalid JWT signature.');
        }

        $payload = json_decode(self::base64UrlDecode($encodedPayload), true, 512, JSON_THROW_ON_ERROR);

        if (isset($payload['exp']) && $payload['exp'] < time()) {
            throw new \RuntimeException('JWT token has expired.');
        }

        return $payload;
    }
}
?>

PHP Standards Recommendations (FIG): The standards that unite modern PHP frameworks.

<?php
/**
 * Modern PHP Standards Reference (PHP FIG):
 * -----------------------------------------------------------------------------
 * PSR-1  : Basic Coding Standard (tags, encoding, namespaces, class & method naming)
 * PSR-4  : Autoloading Standard (maps namespaces directly to file system directories)
 *          e.g. App\\Controllers\\UserController => src/Controllers/UserController.php
 * PSR-7  : HTTP Message Interfaces (ServerRequestInterface, ResponseInterface, UriInterface)
 *          Immutable value objects representing HTTP requests and responses.
 * PSR-11 : Container Interface (ContainerInterface::get(), ContainerInterface::has())
 *          Unified contract for Dependency Injection Containers used in Laravel & Symfony.
 * PSR-12 : Extended Coding Style Guide (standardized formatting, strict type placement, braces)
 * PSR-15 : HTTP Server Request Handlers & Middleware (MiddlewareInterface::process())
 * -----------------------------------------------------------------------------
 */

use Psr\Http\Message\ServerRequestInterface;
use Psr\Http\Message\ResponseInterface;
use Psr\Http\Server\MiddlewareInterface;
use Psr\Http\Server\RequestHandlerInterface;

// Example PSR-15 Compliant Middleware
class AuthMiddleware implements MiddlewareInterface
{
    public function process(ServerRequestInterface $request, RequestHandlerInterface $handler): ResponseInterface
    {
        $authHeader = $request->getHeaderLine('Authorization');
        if (empty($authHeader) || !str_starts_with($authHeader, 'Bearer ')) {
            // Return 401 Unauthorized PSR-7 Response
            return new \Nyholm\Psr7\Response(401, ['Content-Type' => 'application/json'], json_encode(['error' => 'Unauthorized']));
        }

        // Delegate to next middleware / controller in the pipeline
        return $handler->handle($request);
    }
}
?>

Composer Production Optimization: Classmaps, Dev/Prod flags & memory tuning.

{
    "name": "enterprise/core-php-api",
    "description": "Production REST API built with modern Core PHP",
    "type": "project",
    "require": {
        "php": "^8.2",
        "vlucas/phpdotenv": "^5.5",
        "psr/http-message": "^2.0"
    },
    "require-dev": {
        "phpunit/phpunit": "^10.0",
        "phpstan/phpstan": "^1.10"
    },
    "autoload": {
        "psr-4": {
            "App\\": "src/"
        },
        "files": [
            "src/helpers.php"
        ]
    },
    "scripts": {
        "optimize": [
            "composer dump-autoload --optimize --classmap-authoritative --no-dev"
        ]
    }
}
⚡ Production Performance Tip: Running composer dump-autoload -o -a --no-dev pre-computes an associative array of all class-to-file locations. This achieves O(1) instant lookup time and prevents filesystem disk scans on every single web request.
