Modern PHPUnit Architecture: PHP 8 Attributes (#[Test], #[DataProvider]) and Strict Assertions.

<?php
declare(strict_types=1);

namespace Tests\Unit;

use PHPUnit\Framework\TestCase;
use PHPUnit\Framework\Attributes\Test;
use PHPUnit\Framework\Attributes\DataProvider;
use PHPUnit\Framework\Attributes\TestDox;
use App\Services\PricingCalculator;

final class PricingCalculatorTest extends TestCase
{
    private PricingCalculator $calculator;

    protected function setUp(): void
    {
        // Fresh fixture instantiated before EVERY individual test method
        $this->calculator = new PricingCalculator(taxRate: 0.18);
    }

    #[Test]
    #[TestDox('Calculates gross price with standard 18% GST tax rate')]
    public function it_calculates_gross_price_correctly(): void
    {
        $net = 1000.0;
        $gross = $this->calculator->calculateGross($net);

        $this->assertSame(1180.0, $gross);
    }

    #[Test]
    #[DataProvider('discountScenarioProvider')]
    public function it_applies_tier_discounts_accurately(float $subtotal, float $discountPercent, float $expected): void
    {
        $final = $this->calculator->applyDiscount($subtotal, $discountPercent);
        $this->assertEqualsWithDelta($expected, $final, 0.001);
    }

    public static function discountScenarioProvider(): array
    {
        return [
            'No discount below threshold'  => [500.0, 0.0, 500.0],
            '10% Silver customer discount' => [1000.0, 10.0, 900.0],
            '25% Black Friday promotion'   => [2000.0, 25.0, 1500.0],
        ];
    }
}
?>

Test Doubles & Mocking: Isolating unit tests from external I/O, database, and email servers.

<?php
declare(strict_types=1);

namespace Tests\Unit;

use PHPUnit\Framework\TestCase;
use App\Services\OrderService;
use App\Contracts\PaymentGatewayInterface;
use App\Contracts\MailerInterface;
use App\Models\Order;

final class OrderServiceTest extends TestCase
{
    public function test_order_checkout_charges_gateway_and_dispatches_email(): void
    {
        // 1. Create Mock for Payment Gateway:
        $gatewayMock = $this->createMock(PaymentGatewayInterface::class);
        $gatewayMock->expects($this->once())
            ->method('charge')
            ->with(1500, 'cus_test_9921')
            ->willReturn(['status' => 'succeeded', 'txn_id' => 'tx_98124']);

        // 2. Create Mock for Mailer:
        $mailerMock = $this->createMock(MailerInterface::class);
        $mailerMock->expects($this->once())
            ->method('sendReceipt')
            ->with('alice@example.com', 'tx_98124');

        // 3. Inject mocks into OrderService:
        $orderService = new OrderService(paymentGateway: $gatewayMock, mailer: $mailerMock);

        $order = new Order(id: 101, amount: 1500, customerId: 'cus_test_9921', email: 'alice@example.com');
        $result = $orderService->processCheckout($order);

        $this->assertTrue($result->isSuccessful());
    }
}
?>

SOLID Principles & PSR-11 Dependency Injection: Loose coupling for high testability.

<?php
declare(strict_types=1);

namespace App\Core;

use Psr\Container\ContainerInterface;
use ReflectionClass;

// Minimalistic PSR-11 Compliant Auto-Wiring Container:
class Container implements ContainerInterface
{
    private array $bindings = [];
    private array $instances = [];

    public function bind(string $abstract, string|callable $concrete): void
    {
        $this->bindings[$abstract] = $concrete;
    }

    public function get(string $id): mixed
    {
        if (isset($this->instances[$id])) {
            return $this->instances[$id];
        }

        $concrete = $this->bindings[$id] ?? $id;

        if (is_callable($concrete)) {
            return $concrete($this);
        }

        $reflector = new ReflectionClass($concrete);
        $constructor = $reflector->getConstructor();

        if ($constructor === null) {
            return new $concrete();
        }

        // Auto-resolve constructor parameter dependencies!
        $dependencies = [];
        foreach ($constructor->getParameters() as $param) {
            $type = $param->getType();
            if ($type instanceof \ReflectionNamedType && !$type->isBuiltin()) {
                $dependencies[] = $this->get($type->getName());
            }
        }

        return $reflector->newInstanceArgs($dependencies);
    }

    public function has(string $id): bool
    {
        return isset($this->bindings[$id]) || class_exists($id);
    }
}
?>
