Production Axios Client Setup (Auto-CSRF & 401 Handling)

Standard HTTP client for decoupled SPAs connecting to Laravel 11: 

import axios from 'axios';

const api = axios.create({
    baseURL: import.meta.env.VITE_API_URL || 'http://localhost:8000',
    withCredentials: true,
    withXSRFToken: true,
    headers: { 'Accept': 'application/json' },
});

api.interceptors.response.use(
    response => response,
    async error => {
        if (error.response?.status === 401) {
            window.location.href = '/login';
        }
        if (error.response?.status === 419) {
            await axios.get(`${api.defaults.baseURL}/sanctum/csrf-cookie`, { withCredentials: true });
            return api(error.config); // Retry original request
        }
        return Promise.reject(error);
    }
);

export default api;

Laravel Echo + Laravel Reverb WebSocket Template

Complete React hook integration with private channel authorization and automatic teardown: 

import React, { useState, useEffect } from 'react';
import { echo } from '@/echo';

export function LiveChat({ roomId }) {
    const [messages, setMessages] = useState([]);

    useEffect(() => {
        const channelName = `chat.room.${roomId}`;

        // 1. Subscribe to channel:
        echo.private(channelName)
            .listen('.NewChatMessage', (e) => {
                setMessages(prev => [...prev, e.message]);
            });

        // 2. MANDATORY CLEANUP on unmount or roomId change:
        return () => {
            echo.leave(channelName);
        };
    }, [roomId]);

    return (
        <div>
            {messages.map(m => <div key={m.id}>{m.text}</div>)}
        </div>
    );
}
React Error Boundary — Preventing White Screen of Death

Catch runtime rendering exceptions and display a graceful recovery interface: 

export class ErrorBoundary extends React.Component {
    state = { hasError: false };

    static getDerivedStateFromError(error) {
        return { hasError: true };
    }

    componentDidCatch(error, info) {
        console.error("Crash logged to Sentry:", error, info);
    }

    render() {
        if (this.state.hasError) {
            return (
                <div className="alert-box">
                    <h3>Application Encountered an Error</h3>
                    <button onClick={() => window.location.reload()}>
                        Reload Application
                    </button>
                </div>
            );
        }
        return this.props.children;
    }
}
