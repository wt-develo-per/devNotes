The Inertia useForm() Helper — Ultimate Developer Experience

Eliminates 50 lines of boilerplate form management, CSRF tokens, file uploads, and 422 error bindings: 

import { useForm } from '@inertiajs/react';

export default function EditUser({ user }) {
    const { data, setData, put, processing, errors, isDirty } = useForm({
        name: user.name,
        email: user.email,
        avatar: null,
    });

    const submit = (e) => {
        e.preventDefault();
        put(`/users/${user.id}`, {
            preserveScroll: true,
            onSuccess: () => alert('User updated!'),
        });
    };

    return (
        <form onSubmit={submit}>
            <input value={data.name} onChange={e => setData('name', e.target.value)} />
            {errors.name && <span className="error">{errors.name}</span>}
            
            <button disabled={processing || !isDirty}>
                {processing ? 'Saving...' : 'Save Changes'}
            </button>
        </form>
    );
}
HandleInertiaRequests Middleware & usePage()

Share authenticated user info, permissions, and flash alerts across all React views without prop drilling:

HandleInertiaRequests.php: 
public function share(Request $request): array
{
    return array_merge(parent::share($request), [
        'auth.user' => fn () => $request->user() 
            ? $request->user()->only('id', 'name', 'email')
            : null,
        'flash' => [
            'success' => fn () => $request->session()->get('success'),
        ],
    ]);
}
React Component (usePage()): 
import { usePage } from '@inertiajs/react';

export default function TopNav() {
    const { auth, flash } = usePage().props;

    return (
        <header>
            <span>Logged in as: {auth.user?.name}</span>
            {flash.success && (
                <div className="toast">{flash.success}</div>
            )}
        </header>
    );
}

Persistent Layouts — Zero Remounting on Navigation

Keep your navigation bar, sidebar scroll position, and audio playback alive across page transitions: 

// resources/js/Pages/Orders/Index.jsx
import DashboardLayout from '@/Layouts/DashboardLayout';

export default function OrdersIndex({ orders }) {
    return <div>Orders Table Content</div>;
}

// Attach persistent layout function:
OrdersIndex.layout = (page) => <DashboardLayout title="Orders">{page}</DashboardLayout>;

// Result:
// When navigating from /orders to /users:
// DashboardLayout DOES NOT remount! Only the child {page} swaps out!
