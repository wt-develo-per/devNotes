Multi-Scan Controller vs Single-Scan Action

❌ Multi-Scan Controller (8 Queries, 3.8s Latency)

public function index() {
    $total = Order::count();
    $revenue = Order::where('payment_status', 'paid')->sum('total');
    $pending = Order::where('status', 'pending')->count();
    $shipped = Order::where('status', 'shipped')->count();
    $today = Order::whereDate('created_at', today())->sum('total');
    // Repeated table scans overload database CPU!
}

✅ Single-Scan Aggregation (1 Query, 14ms Latency)

public function index() {
    $kpis = DB::table('orders')->selectRaw("
        COUNT(*) AS total_orders,
        SUM(CASE WHEN payment_status = 'paid' THEN total_amount ELSE 0 END) AS gross_revenue,
        COUNT(CASE WHEN status = 'pending' THEN 1 END) AS pending_orders,
        SUM(CASE WHEN created_at >= CURDATE() AND payment_status = 'paid' THEN total_amount ELSE 0 END) AS today_revenue
    ")->first();
    // 1 single scan calculates all registers in memory!
}
