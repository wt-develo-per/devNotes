Leaky Model Broadcast vs Sanitized Enterprise Event

❌ Leaky Event (Exposes Raw Eloquent Model to Public Web)

class OrderUpdated implements ShouldBroadcast {
    // Exposes entire order object including password hashes,
    // billing addresses, and raw vendor commission calculations!
    public $order;
    
    public function broadcastOn() {
        return new Channel('orders'); // Public channel! Anyone can listen!
    }
}

✅ Sanitized Enterprise Event (Private Channels + Minimal Payload)

class OrderStatusUpdatedEvent implements ShouldBroadcast {
    public function __construct(public Order $order) {
        $this->onQueue('high');
    }

    public function broadcastOn(): array {
        // Strict user-isolated private channel
        return [new PrivateChannel("orders.{$this->order->id}")];
    }

    public function broadcastWith(): array {
        // Whitelist exact fields sent over the wire
        return [
            'order_id' => $this->order->id,
            'status' => $this->order->status,
            'tracking_number' => $this->order->tracking_number,
        ];
    }
}

Zombie Listener Memory Leak vs Self-Cleaning Custom Hook

❌ Unmanaged Echo Listener (Zombie Listeners on Every Route Change)

useEffect(() => {
    // Listens to channel but NEVER stops listening on unmount!
    window.Echo.private(`orders.${orderId}`)
        .listen('.order.updated', (e) => setOrder(e));
    // Consequence: When user revisits page, 2x, 3x duplicate audio chimes!
}, []);

✅ Self-Cleaning Hook (Zero Memory Leaks)

export function useEchoPrivateChannel(channelName, eventName, callback) {
    const callbackRef = useRef(callback);
    callbackRef.current = callback;

    useEffect(() => {
        const channel = window.Echo.private(channelName);
        channel.listen(`.${eventName}`, (data) => callbackRef.current(data));

        return () => {
            // Clean teardown prevents duplicate listeners
            channel.stopListening(`.${eventName}`);
            window.Echo.leave(channelName);
        };
    }, [channelName, eventName]);
}
