Naive Migration vs Production-Hardened Indexed Schema

❌ Naive E-Commerce Migration (High Latency & Deadlocks)

Schema::create('products', function (Blueprint $table) {
    $table->id();
    $table->integer('vendor_id'); // Missing FK constraint!
    $table->integer('category_id'); // No index! Full table scans!
    $table->string('title');
    $table->string('price'); // Storing money as varchar or float!
    $table->integer('stock'); // Single stock field = No SKU variants!
    $table->timestamps();
});

✅ Production-Hardened Schema (Composite Indexes & FK Constraints)

Schema::create('products', function (Blueprint $table) {
    $table->id();
    $table->foreignId('vendor_id')->constrained('vendor_profiles')->cascadeOnDelete();
    $table->foreignId('category_id')->constrained('categories')->restrictOnDelete();
    $table->string('title', 255);
    $table->string('slug', 270)->unique();
    $table->decimal('base_price', 10, 2); // Exact precision decimal
    $table->boolean('is_active')->default(true);
    $table->timestamps();

    // Composite index for instant faceted catalog filters
    $table->index(['category_id', 'is_active', 'base_price']);
    $table->fullText(['title']);
});
