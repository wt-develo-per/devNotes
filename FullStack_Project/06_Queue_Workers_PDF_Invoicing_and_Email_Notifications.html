Synchronous Controller Code vs Production Queued Job

❌ Synchronous Controller (Blocks Web Worker)

public function completeOrder(Order $order) {
    // 1. Blocks user for 3 seconds while rendering PDF
    $pdf = Pdf::loadView('invoice', compact('order'));
    
    // 2. Blocks user for another 2 seconds on SMTP network call
    Mail::to($order->email)->send(new InvoiceMail($pdf));
    
    // Total latency: 5+ seconds!
    return redirect()->route('order.success');
}

✅ Asynchronous Event + Job (15ms Instant Return)

public function completeOrder(Order $order) {
    // 1. Marks order paid in DB transaction
    $order->update(['payment_status' => 'paid']);

    // 2. Dispatches event and immediately returns!
    OrderPaymentCompleted::dispatch($order);

    // Total latency: 15 milliseconds!
    return redirect()->route('order.success');
}
// Listener runs in background: GenerateOrderInvoicePdfJob::dispatch($order)->onQueue('invoices');
