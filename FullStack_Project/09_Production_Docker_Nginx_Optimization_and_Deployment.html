Default Nginx vs Production Hardened WebSocket Configuration

❌ Naive Nginx (Breaks WebSockets & Leaks Headers)

server {
    listen 80;
    server_name example.com;
    root /var/www/html; // Exposes .env, .git, and composer.json!
    
    location ~ \.php$ {
        fastcgi_pass 127.0.0.1:9000;
        include fastcgi_params;
    }
    // Missing WebSockets proxy! Reverb connections fail with HTTP 400!
}

✅ Production Hardened Nginx (WebSocket Upgrade + Asset Cache)

server {
    listen 80;
    root /var/www/html/public; // Strict public/ docroot isolation
    server_tokens off;        // Hides Nginx version from attackers

    // WebSockets reverse proxy for Laravel Reverb
    location /app {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "Upgrade";
    }

    // Aggressive immutable cache for compiled Vite bundles
    location ~* \.(css|js|webp|png)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
}
