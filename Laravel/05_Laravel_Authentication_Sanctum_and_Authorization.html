Model Policy: Granular ownership rules for CRUD operations on Eloquent models.

<?php
namespace App\Policies;

use App\Models\Invoice;
use App\Models\User;
use Illuminate\Auth\Access\Response;

final class InvoicePolicy
{
    /**
     * Determine whether the user can view the invoice.
     */
    public function view(User $user, Invoice $invoice): Response
    {
        return $user->id === $invoice->user_id || $user->isSupportStaff()
            ? Response::allow()
            : Response::deny('You do not own this billing invoice.');
    }

    /**
     * Determine whether the user can update the invoice.
     */
    public function update(User $user, Invoice $invoice): bool
    {
        // Unpaid invoices can only be modified if less than 30 days old
        return $user->id === $invoice->user_id 
            && $invoice->status === 'unpaid' 
            && $invoice->created_at->gt(now()->subDays(30));
    }
}

// In Controller: 1-line authorization check:
public function update(UpdateInvoiceRequest $request, Invoice $invoice)
{
    $this->authorize('update', $invoice); // Throws 403 HttpException if denied!
    // Safe execution...
}
?>

Gates & SuperAdmin Master Bypass in AppServiceProvider.

<?php
namespace App\Providers;

use App\Models\User;
use Illuminate\Support\Facades\Gate;
use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        // 1. SuperAdmin Master Bypass:
        // Gate::before() runs BEFORE any Policy or Gate check!
        Gate::before(function (User $user, string $ability) {
            if ($user->hasRole('super-admin')) {
                return true; // Bypass all checks across the entire system!
            }
        });

        // 2. Closure-Based Standalone Gates:
        Gate::define('access-pulse-dashboard', function (User $user) {
            return in_array($user->email, config('app.pulse_allowed_users', []));
        });

        Gate::define('manage-billing', function (User $user, \App\Models\Team $team) {
            return $user->id === $team->owner_id;
        });
    }
}
?>

Multi-Authentication: Separate DB tables, sessions, and guards for Admin vs Customer.

<?php
// config/auth.php
return [
    'guards' => [
        'web' => [
            'driver'   => 'session',
            'provider' => 'users',
        ],
        'admin' => [
            'driver'   => 'session',
            'provider' => 'admins',
        ],
        'api' => [
            'driver'   => 'sanctum',
            'provider' => 'users',
        ],
    ],

    'providers' => [
        'users' => [
            'driver' => 'eloquent',
            'model'  => \App\Models\User::class,
        ],
        'admins' => [
            'driver' => 'eloquent',
            'model'  => \App\Models\Admin::class,
        ],
    ],
];

// Controller login for admin:
Auth::guard('admin')->attempt(['email' => $email, 'password' => $pass]);

// Route protection:
Route::middleware('auth:admin')->prefix('admin')->group(function () {
    Route::get('/dashboard', [AdminDashboardController::class, 'index']);
});
?>
