Flexible Caching (Stale-While-Revalidate): Users never wait for cache warmups!

<?php
use Illuminate\Support\Facades\Cache;

// 🚀 Modern Laravel 11, 12 & 13 Feature:
// Syntax: Cache::flexible('key', [freshSeconds, staleSeconds], callback)

$topProducts = Cache::flexible('leaderboard:products', [300, 900], function () {
    // Heavy aggregated analytical query (takes 1,200ms)
    return DB::table('order_items')
        ->select('product_id', DB::raw('SUM(quantity) as sold'))
        ->groupBy('product_id')
        ->orderByDesc('sold')
        ->limit(10)
        ->get();
});

// Under the hood:
// 0 - 5 min: Data is fresh, served instantly from Redis.
// 5 - 15 min: Data is slightly stale. First user STILL gets instantaneous response (0.5ms),
//             while a background queue job warms up fresh data for subsequent users!
// 15+ min: Data completely expired, recalculates synchronously.
?>

Redis Cache Tagging: Flush an entire department or tenant cache in 1 millisecond.

<?php
use Illuminate\Support\Facades\Cache;

// 1. Tagging keys with multiple semantic labels (Redis/Memcached only!)
$companyId = 42;
$userId = 109;

Cache::tags(["company_{$companyId}", 'users', 'billing'])->remember(
    "user_{$userId}_billing_summary",
    now()->addHours(6),
    function () use ($userId) {
        return Invoice::where('user_id', $userId)->with('items')->get();
    }
);

// 2. Scenario A: Single user updates billing -> flush their specific key:
Cache::tags(["company_{$companyId}"])->forget("user_{$userId}_billing_summary");

// 3. Scenario B: Company changes subscription tier -> flush ALL company keys at once!
Cache::tags(["company_{$companyId}"])->flush();
// Every cached query belonging to company_42 is wiped instantly, while other companies remain cached!
?>

Distributed Atomic Locks: Stop concurrent double-payouts and race conditions across multiple server nodes.

<?php
use Illuminate\Support\Facades\Cache;

$walletId = 8812;
$lockKey = "wallet-withdrawal-lock:{$walletId}";

// Acquire atomic lock for 10 seconds:
$lock = Cache::lock($lockKey, 10);

if ($lock->get()) {
    try {
        // Exclusive Critical Section:
        // No other request or server can enter this block for wallet #8812!
        $wallet = Wallet::findOrFail($walletId);
        
        if ($wallet->balance >= $amount) {
            $wallet->decrement('balance', $amount);
            PaymentGateway::disburseFunds($wallet->user, $amount);
        }
    } finally {
        // Always release lock inside finally block!
        $lock->release();
    }
} else {
    // Another concurrent request is already processing this wallet!
    abort(429, 'Another withdrawal request is currently in flight. Please wait.');
}

// Alternative blocking syntax (wait up to 5 seconds to acquire lock):
Cache::lock('daily-report-generation', 60)->block(5, function () {
    // Guaranteed single-process execution!
});
?>
