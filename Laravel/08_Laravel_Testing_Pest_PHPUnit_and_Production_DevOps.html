Pest v3: Elegant functional syntax + Architectural Rules enforcement.

<?php
// tests/Feature/OrderCreationTest.php
use App\Models\User;
use App\Models\Product;

it('allows authenticated users to purchase in-stock products', function () {
    // 1. Arrange: Factory creation
    $user = User::factory()->create();
    $product = Product::factory()->create(['stock' => 5, 'price' => 2000]);

    // 2. Act: Authenticated HTTP Post
    $response = $this->actingAs($user)->postJson('/api/v1/orders', [
        'items' => [
            ['id' => $product->id, 'qty' => 2]
        ]
    ]);

    // 3. Assert: Expectations
    $response->assertCreated()
        ->assertJsonPath('data.total_amount', 4000);

    expect($product->fresh()->stock)->toBe(3);
});

// 🛡️ Pest Architecture Test (Guarantees team adheres to Clean Architecture!):
test('controllers do not directly access database queries')
    ->expect('App\Http\Controllers')
    ->not->toUse(['Illuminate\Support\Facades\DB', 'Illuminate\Database\Eloquent\Builder'])
    ->ignoring('App\Http\Controllers\RawReportExportController');
?>

HTTP Feature Testing: Database transactions + Model Factory recycling.

<?php
namespace Tests\Feature;

use App\Models\Invoice;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

final class InvoiceApiTest extends TestCase
{
    // Transactional rollback after every test method:
    use RefreshDatabase;

    public function test_user_cannot_view_another_users_invoice(): void
    {
        // Arrange: 2 distinct users
        $owner = User::factory()->create();
        $intruder = User::factory()->create();

        $invoice = Invoice::factory()->for($owner)->create([
            'amount' => 15000,
            'status' => 'paid',
        ]);

        // Act: Intruder tries to read owner's invoice
        $response = $this->actingAs($intruder)->getJson("/api/v1/invoices/{$invoice->id}");

        // Assert: 403 Forbidden
        $response->assertForbidden();
    }
}
?>

Service Fakes: 100% test isolation — Zero network calls, zero external SMS/Emails!

<?php
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Facades\Queue;
use App\Mail\OrderShippedMail;
use App\Jobs\SyncWithWarehouseJob;

test('order completion dispatches email, queue job, and calls logistics webhook', function () {
    // 1. Fake all external systems
    Http::fake([
        'api.logistics.com/*' => Http::response(['tracking_number' => 'FEDEX-9988'], 200),
    ]);
    Mail::fake();
    Queue::fake();

    $order = Order::factory()->create(['status' => 'processing']);

    // 2. Trigger Action
    $action = app(\App\Actions\CompleteOrderAction::class);
    $action->execute($order);

    // 3. Assert Fakes
    Mail::assertQueued(OrderShippedMail::class, fn($mail) => $mail->hasTo($order->customer_email));
    Queue::assertPushed(SyncWithWarehouseJob::class);
    Http::assertSent(fn($req) => $req->url() === 'https://api.logistics.com/v1/shipments');
});
?>
