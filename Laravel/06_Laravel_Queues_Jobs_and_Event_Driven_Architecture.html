Job Chaining: Strict sequential order — Next job only executes if previous succeeds!

<?php
use Illuminate\Support\Facades\Bus;
use App\Jobs\GeneratePdfInvoice;
use App\Jobs\ChargeCustomerCard;
use App\Jobs\SendReceiptEmail;

// Strict dependency pipeline:
Bus::chain([
    new ChargeCustomerCard($order),     // Step 1: Charge card
    new GeneratePdfInvoice($order),     // Step 2: Only runs if charge succeeded
    new SendReceiptEmail($order),       // Step 3: Only runs if PDF generated
])->catch(function (\Throwable $e) use ($order) {
    // Pipeline failed at any step!
    \Log::channel('payments')->critical("Order pipeline failed for #{$order->id}: " . $e->getMessage());
    $order->update(['status' => 'payment_failed']);
})->dispatch();
?>

Job Batching: Process 10,000 items in parallel across 20 workers with live progress tracking.

<?php
use Illuminate\Support\Facades\Bus;
use Illuminate\Bus\Batch;
use App\Jobs\ProcessCsvRow;

$jobs = $csvRows->map(fn($row) => new ProcessCsvRow($row));

$batch = Bus::batch($jobs)
    ->name('Importing 50,000 Catalog Products')
    ->onQueue('imports')
    ->allowFailures() // Don't abort entire import if 2 rows fail!
    ->then(function (Batch $batch) {
        // All 50,000 jobs completed successfully!
        NotifyUserImportFinished::dispatch($batch->id);
    })
    ->catch(function (Batch $batch, Throwable $e) {
        // First batch failure detected
    })
    ->finally(function (Batch $batch) {
        // Runs when batch finishes, regardless of failure count
        CleanupTempFiles::dispatch($batch->id);
    })
    ->dispatch();

// Frontend polls batch progress API:
return response()->json([
    'batch_id' => $batch->id,
    'progress' => $batch->progress(), // e.g. 64%
    'total'    => $batch->totalJobs,
    'failed'   => $batch->failedJobs,
]);
?>

Laravel 12/13 Concurrency Facade: Run async tasks in parallel without queue overhead!

<?php
// 🚀 Available in Laravel 12 and Laravel 13:
use Illuminate\Support\Facades\Concurrency;

// Problem: Calling 3 third-party APIs sequentially takes 600ms + 400ms + 500ms = 1,500ms!
// Solution: Run all 3 concurrently using PHP Forks or Fibers:

[$exchangeRates, $shippingQuotes, $fraudScore] = Concurrency::run([
    fn() => Http::timeout(2)->get('https://api.fx.com/rates')->json(),
    fn() => Http::timeout(2)->get('https://api.fedex.com/quotes')->json(),
    fn() => Http::timeout(2)->post('https://api.fraud.com/check', ['ip' => $ip])->json(),
]);

// Total response time = Max single call (600ms instead of 1,500ms — 60% faster!)
return response()->json([
    'rates'    => $exchangeRates,
    'shipping' => $shippingQuotes,
    'fraud'    => $fraudScore,
]);
?>
