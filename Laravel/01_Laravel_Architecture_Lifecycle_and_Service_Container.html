Anti-Pattern: Business logic, validation, payment gateway, and email mixed inside controller.

<?php
namespace App\Http\Controllers;

use App\Models\User;
use App\Models\Order;
use Illuminate\Http\Request;
use Stripe\StripeClient;
use Illuminate\Support\Facades\Mail;

class OrderController extends Controller
{
    // ❌ ANTI-PATTERN: Fat Controller with 5 distinct responsibilities
    public function store(Request $request)
    {
        // 1. Manual ad-hoc validation
        $validated = $request->validate([
            'user_id' => 'required|exists:users,id',
            'amount'   => 'required|numeric|min:1',
            'token'    => 'required|string',
        ]);

        // 2. Hardcoded tight coupling (Cannot mock in unit tests!)
        $stripe = new StripeClient(config('services.stripe.secret'));
        $charge = $stripe->charges->create([
            'amount'   => $validated['amount'] * 100,
            'currency' => 'usd',
            'source'   => $validated['token'],
        ]);

        // 3. Direct DB mutation without transaction
        $order = Order::create([
            'user_id'    => $validated['user_id'],
            'amount'     => $validated['amount'],
            'charge_id'  => $charge->id,
            'status'     => 'paid',
        ]);

        // 4. Synchronous email (Blocks the HTTP request!)
        Mail::raw("Your order #{$order->id} is confirmed!", function ($msg) use ($request) {
            $msg->to($request->user()->email)->subject("Order Paid");
        });

        return response()->json($order, 201);
    }
}
?>

Senior Architecture: FormRequest + Single-Action Class + DB Transaction + Dispatched Event.

<?php
namespace App\Actions\Orders;

use App\Contracts\PaymentGatewayInterface;
use App\Data\OrderData;
use App\Events\OrderPlaced;
use App\Models\Order;
use Illuminate\Support\Facades\DB;

final readonly class CreateOrderAction
{
    // ✅ Dependency Inversion: Interface injected via Container
    public function __construct(
        private PaymentGatewayInterface $paymentGateway
    ) {}

    public function execute(OrderData $data): Order
    {
        return DB::transaction(function () use ($data) {
            // 1. Charge through decoupled gateway interface
            $charge = $this->paymentGateway->charge(
                cents: $data->amountInCents,
                token: $data->paymentToken
            );

            // 2. Atomic Order Persistence
            $order = Order::create([
                'user_id'        => $data->userId,
                'total_amount'   => $data->amount,
                'transaction_id' => $charge->id,
                'status'         => 'paid',
            ]);

            // 3. Event-Driven: Emails, Notifications, and Invoicing handled asynchronously
            OrderPlaced::dispatch($order);

            return $order;
        });
    }
}

// In the Controller: Ultra slim, zero business logic!
namespace App\Http\Controllers;

use App\Actions\Orders\CreateOrderAction;
use App\Http\Requests\CreateOrderRequest;
use Illuminate\Http\JsonResponse;

final class OrderStoreController
{
    public function __invoke(CreateOrderRequest $request, CreateOrderAction $action): JsonResponse
    {
        $order = $action->execute($request->toOrderData());

        return response()->json(['order_id' => $order->id], 201);
    }
}
?>

Service Provider: Binding Interface to Concrete Implementation & Contextual Swapping.

<?php
namespace App\Providers;

use App\Contracts\PaymentGatewayInterface;
use App\Services\StripePaymentGateway;
use App\Services\PayPalPaymentGateway;
use Illuminate\Support\ServiceProvider;

final class PaymentServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        // 1. Standard Singleton Binding:
        $this->app->singleton(PaymentGatewayInterface::class, function ($app) {
            return new StripePaymentGateway(
                secretKey: config('services.stripe.secret'),
                webhookSecret: config('services.stripe.webhook_secret')
            );
        });

        // 2. Contextual Binding: If a specific controller needs PayPal instead
        $this->app->when(\App\Http\Controllers\PayPalWebhookController::class)
            ->needs(PaymentGatewayInterface::class)
            ->give(PayPalPaymentGateway::class);
    }
}
?>
