Laravel 11, 12 & 13 Modern Architecture: Fluid chaining in bootstrap/app.php.

<?php
// bootstrap/app.php (Modern Laravel 11 / 12 / 13)
use Illuminate\Foundation\Application;
use Illuminate\Foundation\Configuration\Middleware;

return Application::configure(basePath: dirname(__DIR__))
    ->withRouting(
        web: __DIR__.'/../routes/web.php',
        api: __DIR__.'/../routes/api.php',
        commands: __DIR__.'/../routes/console.php',
    )
    ->withMiddleware(function (Middleware $middleware) {
        // 1. Global Middleware: append ya prepend karein
        $middleware->append(\App\Http\Middleware\SecurityHeadersMiddleware::class);

        // 2. Route Middleware Aliases define karein
        $middleware->alias([
            'role'    => \App\Http\Middleware\CheckRole::class,
            'tenant'  => \App\Http\Middleware\IdentifyTenant::class,
        ]);

        // 3. Web / API Group customization
        $middleware->api(prepend: [
            \App\Http\Middleware\ForceJsonRequest::class,
        ]);

        // 4. Exclude specific routes from CSRF
        $middleware->validateCsrfTokens(except: [
            'stripe/webhook',
            'api/v1/payments/razorpay-callback',
        ]);

        // 5. Middleware Priority / Sorting
        $middleware->priority([
            \Illuminate\Session\Middleware\StartSession::class,
            \Illuminate\View\Middleware\ShareErrorsFromSession::class,
            \App\Http\Middleware\Authenticate::class,
            \Illuminate\Routing\Middleware\SubstituteBindings::class,
        ]);
    })->create();
?>

Laravel 11+ Controller Middleware: $this->middleware() constructor deprecated! Use HasMiddleware.

<?php
namespace App\Http\Controllers;

use App\Models\Invoice;
use Illuminate\Routing\Controllers\HasMiddleware;
use Illuminate\Routing\Controllers\Middleware;
use Illuminate\Http\Request;

final class InvoiceController extends Controller implements HasMiddleware
{
    /**
     * ✅ Modern Laravel 11/12/13 Standard:
     * Static middleware definition with granular 'only' and 'except' filters.
     */
    public static function middleware(): array
    {
        return [
            // Har method par chalega
            'auth:sanctum',

            // Sirf sensitive actions par chalega
            new Middleware('subscribed', only: ['downloadPdf', 'export']),

            // Admin verification sirf delete aur refund par
            new Middleware('role:admin', only: ['destroy', 'refund']),

            // Public method ko bypass karne ke liye
            new Middleware('throttle:10,1', only: ['store']),
        ];
    }

    public function index(Request $request) { /* ... */ }
    public function show(Invoice $invoice) { /* ... */ }
    public function downloadPdf(Invoice $invoice) { /* ... */ }
    public function destroy(Invoice $invoice) { /* ... */ }
}
?>
